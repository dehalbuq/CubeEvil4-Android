cmake_minimum_required(VERSION 3.16)
project(darkosre_cloudlab_smoke VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
add_executable(cloud-smoke smoke.c)
target_link_libraries(cloud-smoke PRIVATE Threads::Threads)
target_compile_options(cloud-smoke PRIVATE -Wall -Wextra -Werror -O2)
if(BUILD_TESTING)
  add_test(NAME cpu_mmap_wx_64mb COMMAND cloud-smoke 64)
  add_test(NAME rss_readback_256mb COMMAND cloud-smoke 256)
  add_test(NAME rss_readback_512mb COMMAND cloud-smoke 512)
  set_tests_properties(rss_readback_512mb PROPERTIES TIMEOUT 30)
endif()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(EGL QUIET IMPORTED_TARGET egl)
  pkg_check_modules(GLES QUIET IMPORTED_TARGET glesv2)
  if(EGL_FOUND AND GLES_FOUND)
    add_executable(cloud-gles3-smoke gles3-smoke.c)
    target_link_libraries(cloud-gles3-smoke PRIVATE PkgConfig::EGL PkgConfig::GLES)
    target_compile_options(cloud-gles3-smoke PRIVATE -Wall -Wextra -Werror -O2)
    if(BUILD_TESTING)
      add_test(NAME mesa_software_gles3 COMMAND cloud-gles3-smoke)
      set_tests_properties(mesa_software_gles3 PROPERTIES ENVIRONMENT "EGL_PLATFORM=surfaceless;LIBGL_ALWAYS_SOFTWARE=1")
    endif()
  else()
    message(STATUS "EGL/GLES development libraries absent: shader test disabled")
  endif()
endif()
